function matchItems(q,items){
  var ql=q.toLowerCase(),sc=[];
  items.forEach(function(p){
    var s=0;
    var name=(p.web_item_name||p.item_name||p.name||'').toLowerCase();
    var grp=(p.item_group||'').toLowerCase();
    if(name===ql)s+=100;
    else if(name.startsWith(ql))s+=70;
    else if(name.includes(ql))s+=50;
    if(grp===ql)s+=60;
    else if(grp.includes(ql))s+=35;
    q.split(/\s+/).forEach(function(w){
      if(w.length<2)return;
      if(name.includes(w))s+=20;
      if(grp.includes(w))s+=12;
    });
    if(s>0)sc.push({p:p,s:s});
  });
  sc.sort(function(a,b){return b.s-a.s;});
  return sc.slice(0,8).map(function(x){return x.p;});
}
